## IV. Technical Architecture Details

### A. UUE Stage Progression Architecture

#### 1. UueStageProgressionFlow Component
```typescript
interface UueStageProgressionFlowProps {
  currentStage: UueStage;
  stageProgress: UueStageProgress;
  onStageClick: (stage: UueStage) => void;
  onStageTransition: (fromStage: UueStage, toStage: UueStage) => void;
  showRequirements?: boolean;
  showPredictions?: boolean;
}

export default function UueStageProgressionFlow({
  currentStage,
  stageProgress,
  onStageClick,
  onStageTransition,
  showRequirements = true,
  showPredictions = true
}: UueStageProgressionFlowProps) {
  const [isTransitioning, setIsTransitioning] = useState(false);
  const [transitionDirection, setTransitionDirection] = useState<'forward' | 'backward' | null>(null);
  
  const handleStageTransition = async (targetStage: UueStage) => {
    setIsTransitioning(true);
    setTransitionDirection(getTransitionDirection(currentStage, targetStage));
    
    try {
      await onStageTransition(currentStage, targetStage);
      // Trigger transition animation
      setTimeout(() => setIsTransitioning(false), 1000);
    } catch (error) {
      setIsTransitioning(false);
      console.error('Stage transition failed:', error);
    }
  };
  
  return (
    <div className={styles.uueStageProgressionFlow}>
      {/* Stage progression visualization */}
      <div className={styles.stageFlow}>
        {Object.values(UueStage).map((stage, index) => (
          <UueStageNode
            key={stage}
            stage={stage}
            progress={stageProgress[stage.toLowerCase() as keyof UueStageProgress]}
            isCurrent={stage === currentStage}
            isCompleted={isStageCompleted(stage, stageProgress)}
            isAccessible={isStageAccessible(stage, currentStage, stageProgress)}
            onClick={() => onStageClick(stage)}
            onTransition={() => handleStageTransition(stage)}
            showRequirements={showRequirements}
            showPredictions={showPredictions}
          />
        ))}
      </div>
      
      {/* Transition animations */}
      {isTransitioning && (
        <UueStageTransition
          direction={transitionDirection}
          fromStage={currentStage}
          toStage={targetStage}
        />
      )}
    </div>
  );
}
```

#### 2. UueStageRequirements Component
```typescript
interface UueStageRequirementsProps {
  stage: UueStage;
  requirements: StageRequirement[];
  currentProgress: StageProgress;
  onRequirementClick: (requirement: StageRequirement) => void;
}

export default function UueStageRequirements({
  stage,
  requirements,
  currentProgress,
  onRequirementClick
}: UueStageRequirementsProps) {
  const completedRequirements = requirements.filter(req => 
    currentProgress.masteredCriteria.some(c => c.id === req.criterionId)
  );
  
  const remainingRequirements = requirements.filter(req => 
    !completedRequirements.some(c => c.id === req.id)
  );
  
  return (
    <div className={styles.uueStageRequirements}>
      <h3>Requirements for {stage} Stage</h3>
      
      {/* Progress overview */}
      <div className={styles.requirementsProgress}>
        <div className={styles.progressBar}>
          <div
            className={styles.progressFill}
            style={{
              width: `${(completedRequirements.length / requirements.length) * 100}%`
            }}
          />
        </div>
        <span className={styles.progressText}>
          {completedRequirements.length} of {requirements.length} requirements met
        </span>
      </div>
      
      {/* Requirements list */}
      <div className={styles.requirementsList}>
        {requirements.map(requirement => {
          const isCompleted = completedRequirements.some(c => c.id === requirement.id);
          const progress = currentProgress.criteriaProgress.find(c => c.criterionId === requirement.criterionId);
          
          return (
            <div
              key={requirement.id}
              className={`${styles.requirementItem} ${isCompleted ? styles.completed : ''}`}
              onClick={() => onRequirementClick(requirement)}
            >
              <div className={styles.requirementHeader}>
                <span className={styles.requirementTitle}>{requirement.title}</span>
                <span className={styles.requirementStatus}>
                  {isCompleted ? '✅' : '🔄'}
                </span>
              </div>
              
              <div className={styles.requirementDetails}>
                <p className={styles.requirementDescription}>{requirement.description}</p>
                {progress && (
                  <div className={styles.requirementProgress}>
                    <span>Progress: {Math.round(progress.masteryScore * 100)}%</span>
                    {progress.nextReviewAt && (
                      <span>Next Review: {formatDate(progress.nextReviewAt)}</span>
                    )}
                  </div>
                )}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
```

### B. Learning Pathway Discovery Architecture

#### 1. LearningPathwayMap Component
```typescript
interface LearningPathwayMapProps {
  pathways: LearningPathway[];
  currentPosition: PathwayPosition;
  onPathwaySelect: (pathway: LearningPathway) => void;
  onPathwayOptimize: (pathway: LearningPathway) => void;
  showRecommendations?: boolean;
}

export default function LearningPathwayMap({
  pathways,
  currentPosition,
  onPathwaySelect,
  onPathwayOptimize,
  showRecommendations = true
}: LearningPathwayMapProps) {
  const [selectedPathway, setSelectedPathway] = useState<LearningPathway | null>(null);
  const [zoomLevel, setZoomLevel] = useState(1);
  const [panOffset, setPanOffset] = useState({ x: 0, y: 0 });
  
  const handlePathwaySelect = (pathway: LearningPathway) => {
    setSelectedPathway(pathway);
    onPathwaySelect(pathway);
  };
  
  const handlePathwayOptimize = async (pathway: LearningPathway) => {
    try {
      const optimizedPathway = await onPathwayOptimize(pathway);
      // Update pathway with optimization results
      setSelectedPathway(optimizedPathway);
    } catch (error) {
      console.error('Pathway optimization failed:', error);
    }
  };
  
  return (
    <div className={styles.learningPathwayMap}>
      {/* Map controls */}
      <div className={styles.mapControls}>
        <button onClick={() => setZoomLevel(prev => Math.min(prev + 0.1, 2))}>
          Zoom In
        </button>
        <button onClick={() => setZoomLevel(prev => Math.max(prev - 0.1, 0.5))}>
          Zoom Out
        </button>
        <button onClick={() => setPanOffset({ x: 0, y: 0 })}>
          Reset View
        </button>
      </div>
      
      {/* Pathway visualization */}
      <div
        className={styles.pathwayCanvas}
        style={{
          transform: `scale(${zoomLevel}) translate(${panOffset.x}px, ${panOffset.y}px)`
        }}
      >
        {pathways.map(pathway => (
          <PathwayNode
            key={pathway.id}
            pathway={pathway}
            isSelected={selectedPathway?.id === pathway.id}
            currentPosition={currentPosition}
            onClick={() => handlePathwaySelect(pathway)}
            onOptimize={() => handlePathwayOptimize(pathway)}
          />
        ))}
        
        {/* Pathway connections */}
        {pathways.map(pathway => 
          pathway.connections?.map(connection => (
            <PathwayConnection
              key={`${pathway.id}-${connection.targetId}`}
              source={pathway}
              target={pathways.find(p => p.id === connection.targetId)!}
              connectionType={connection.type}
              strength={connection.strength}
            />
          ))
        )}
      </div>
      
      {/* Selected pathway details */}
      {selectedPathway && (
        <div className={styles.pathwayDetails}>
          <h3>{selectedPathway.title}</h3>
          <p>{selectedPathway.description}</p>
          
          <div className={styles.pathwayMetrics}>
            <span>Cost: {selectedPathway.cost}</span>
            <span>Time: {selectedPathway.estimatedTime} min</span>
            <span>Difficulty: {selectedPathway.difficulty}</span>
          </div>
          
          <button
            className={styles.optimizeButton}
            onClick={() => handlePathwayOptimize(selectedPathway)}
          >
            Optimize Pathway
          </button>
        </div>
      )}
    </div>
  );
}
```

### C. Enhanced Mastery Tracking Architecture

#### 1. MasteryHeatmap Component
```typescript
interface MasteryHeatmapProps {
  masteryData: MasteryDataPoint[];
  timeRange: TimeRange;
  onDataPointClick: (dataPoint: MasteryDataPoint) => void;
  showTrends?: boolean;
  showPredictions?: boolean;
}

export default function MasteryHeatmap({
  masteryData,
  timeRange,
  onDataPointClick,
  showTrends = true,
  showPredictions = true
}: MasteryHeatmapProps) {
  const [selectedTimeRange, setSelectedTimeRange] = useState<TimeRange>(timeRange);
  const [heatmapData, setHeatmapData] = useState<HeatmapDataPoint[]>([]);
  
  useEffect(() => {
    // Process mastery data into heatmap format
    const processedData = processMasteryDataForHeatmap(masteryData, selectedTimeRange);
    setHeatmapData(processedData);
  }, [masteryData, selectedTimeRange]);
  
  const handleDataPointClick = (dataPoint: HeatmapDataPoint) => {
    const originalDataPoint = masteryData.find(d => 
      d.date === dataPoint.date && d.criterionId === dataPoint.criterionId
    );
    if (originalDataPoint) {
      onDataPointClick(originalDataPoint);
    }
  };
  
  return (
    <div className={styles.masteryHeatmap}>
      {/* Time range selector */}
      <div className={styles.timeRangeSelector}>
        <select
          value={selectedTimeRange}
          onChange={(e) => setSelectedTimeRange(e.target.value as TimeRange)}
        >
          <option value="7d">Last 7 Days</option>
          <option value="30d">Last 30 Days</option>
          <option value="90d">Last 90 Days</option>
          <option value="1y">Last Year</option>
        </select>
      </div>
      
      {/* Heatmap visualization */}
      <div className={styles.heatmapContainer}>
        {heatmapData.map(dataPoint => (
          <div
            key={`${dataPoint.date}-${dataPoint.criterionId}`}
            className={styles.heatmapCell}
            style={{
              backgroundColor: getHeatmapColor(dataPoint.masteryScore),
              opacity: dataPoint.intensity
            }}
            onClick={() => handleDataPointClick(dataPoint)}
            title={`${dataPoint.criterionTitle}: ${Math.round(dataPoint.masteryScore * 100)}%`}
          />
        ))}
      </div>
      
      {/* Legend */}
      <div className={styles.heatmapLegend}>
        <span>Low Mastery</span>
        <div className={styles.legendGradient} />
        <span>High Mastery</span>
      </div>
      
      {/* Trends and predictions */}
      {showTrends && (
        <div className={styles.trendsSection}>
          <h4>Learning Trends</h4>
          <TrendAnalysis data={masteryData} timeRange={selectedTimeRange} />
        </div>
      )}
      
      {showPredictions && (
        <div className={styles.predictionsSection}>
          <h4>Mastery Predictions</h4>
          <MasteryPredictions data={masteryData} timeRange={selectedTimeRange} />
        </div>
      )}
    </div>
  );
}
```
