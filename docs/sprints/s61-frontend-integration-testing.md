// Virtual scrolling for large section hierarchies
interface VirtualizedSectionTreeProps {
  sections: BlueprintSection[];
  height: number;
  itemHeight: number;
  onSectionSelect: (section: BlueprintSection) => void;
}

export default function VirtualizedSectionTree({
  sections,
  height,
  itemHeight,
  onSectionSelect
}: VirtualizedSectionTreeProps) {
  const [scrollTop, setScrollTop] = useState(0);
  const [containerRef, setContainerRef] = useState<HTMLDivElement | null>(null);
  
  // Calculate visible range
  const visibleCount = Math.ceil(height / itemHeight);
  const startIndex = Math.floor(scrollTop / itemHeight);
  const endIndex = Math.min(startIndex + visibleCount + 1, sections.length);
  
  // Get visible sections
  const visibleSections = sections.slice(startIndex, endIndex);
  
  // Calculate total height and offset
  const totalHeight = sections.length * itemHeight;
  const offsetY = startIndex * itemHeight;
  
  const handleScroll = (event: React.UIEvent<HTMLDivElement>) => {
    setScrollTop(event.currentTarget.scrollTop);
  };
  
  return (
    <div
      ref={setContainerRef}
      className={styles.virtualizedContainer}
      style={{ height, overflow: 'auto' }}
      onScroll={handleScroll}
    >
      <div style={{ height: totalHeight, position: 'relative' }}>
        <div style={{ transform: `translateY(${offsetY}px)` }}>
          {visibleSections.map((section, index) => (
            <SectionTreeItem
              key={section.id}
              section={section}
              depth={section.depth}
              onSelect={() => onSectionSelect(section)}
              style={{ height: itemHeight }}
            />
          ))}
        </div>
      </div>
    </div>
  );
}

// Performance monitoring hook
export function usePerformanceMonitoring(componentName: string) {
  const [renderTime, setRenderTime] = useState(0);
  const [memoryUsage, setMemoryUsage] = useState(0);
  
  useEffect(() => {
    const startTime = performance.now();
    const startMemory = performance.memory?.usedJSHeapSize || 0;
    
    return () => {
      const endTime = performance.now();
      const endMemory = performance.memory?.usedJSHeapSize || 0;
      
      const renderDuration = endTime - startTime;
      const memoryDelta = endMemory - startMemory;
      
      setRenderTime(renderDuration);
      setMemoryUsage(memoryDelta);
      
      // Log performance metrics
      if (renderDuration > 16) { // 60fps threshold
        console.warn(`${componentName} render time: ${renderDuration.toFixed(2)}ms`);
      }
      
      if (memoryDelta > 1024 * 1024) { // 1MB threshold
        console.warn(`${componentName} memory usage: ${(memoryDelta / 1024 / 1024).toFixed(2)}MB`);
      }
      
      // Send to analytics
      analytics.track('component_performance', {
        component: componentName,
        renderTime: renderDuration,
        memoryUsage: memoryDelta
      });
    };
  }, [componentName]);
  
  return { renderTime, memoryUsage };
}

// Usage in components
export default function BlueprintSectionsPage() {
  const { renderTime, memoryUsage } = usePerformanceMonitoring('BlueprintSectionsPage');
  
  // Component logic...
  
  return (
    <div className={styles.blueprintSectionsPage}>
      {/* Performance debug info in development */}
      {process.env.NODE_ENV === 'development' && (
        <div className={styles.performanceDebug}>
          <span>Render: {renderTime.toFixed(2)}ms</span>
          <span>Memory: {(memoryUsage / 1024).toFixed(2)}KB</span>
        </div>
      )}
      
      {/* Component content */}
    </div>
  );
}

```typescript
// Accessible section tree component
interface AccessibleSectionTreeProps {
  sections: BlueprintSection[];
  selectedSection: BlueprintSection | null;
  onSectionSelect: (section: BlueprintSection) => void;
}

export default function AccessibleSectionTree({
  sections,
  selectedSection,
  onSectionSelect
}: AccessibleSectionTreeProps) {
  const [expandedSections, setExpandedSections] = useState<Set<string>>(new Set());
  
  const toggleSection = (sectionId: string) => {
    const newExpanded = new Set(expandedSections);
    if (newExpanded.has(sectionId)) {
      newExpanded.delete(sectionId);
    } else {
      newExpanded.add(sectionId);
    }
    setExpandedSections(newExpanded);
  };
  
  const renderSection = (section: BlueprintSection, depth: number) => {
    const hasChildren = section.children && section.children.length > 0;
    const isExpanded = expandedSections.has(section.id);
    const isSelected = selectedSection?.id === section.id;
    
    return (
      <li
        key={section.id}
        role="treeitem"
        aria-expanded={hasChildren ? isExpanded : undefined}
        aria-selected={isSelected}
        aria-level={depth + 1}
        aria-label={`${section.title}${hasChildren ? `, ${section.children.length} subsections` : ''}`}
      >
        <div className={styles.sectionItem}>
          {hasChildren && (
            <button
              type="button"
              aria-label={`${isExpanded ? 'Collapse' : 'Expand'} ${section.title}`}
              onClick={() => toggleSection(section.id)}
              className={styles.expandButton}
            >
              <span aria-hidden="true">
                {isExpanded ? '▼' : '▶'}
              </span>
            </button>
          )}
          
          <button
            type="button"
            className={`${styles.sectionButton} ${isSelected ? styles.selected : ''}`}
            onClick={() => onSectionSelect(section)}
            aria-current={isSelected ? 'page' : undefined}
          >
            {section.title}
          </button>
          
          {section.masteryProgress && (
            <div
              className={styles.masteryIndicator}
              aria-label={`Mastery progress: ${Math.round(section.masteryProgress.score * 100)}%`}
            >
              <div
                className={styles.masteryBar}
                style={{ width: `${section.masteryProgress.score * 100}%` }}
                aria-hidden="true"
              />
            </div>
          )}
        </div>
        
        {hasChildren && isExpanded && (
          <ul role="group" className={styles.sectionChildren}>
            {section.children.map(child => renderSection(child, depth + 1))}
          </ul>
        )}
      </li>
    );
  };
  
  return (
    <nav aria-label="Blueprint sections navigation">
      <ul role="tree" className={styles.sectionTree}>
        {sections.map(section => renderSection(section, 0))}
      </ul>
    </nav>
  );
}
```

// Keyboard navigation hook
export function useKeyboardNavigation<T>(
  items: T[],
  onSelect: (item: T) => void,
  getItemId: (item: T) => string
) {
  const [focusedIndex, setFocusedIndex] = useState(0);
  const [focusedId, setFocusedId] = useState<string | null>(null);
  
  const handleKeyDown = useCallback((event: KeyboardEvent) => {
    switch (event.key) {
      case 'ArrowDown':
        event.preventDefault();
        setFocusedIndex(prev => Math.min(prev + 1, items.length - 1));
        break;
        
      case 'ArrowUp':
        event.preventDefault();
        setFocusedIndex(prev => Math.max(prev - 1, 0));
        break;
        
      case 'Home':
        event.preventDefault();
        setFocusedIndex(0);
        break;
        
      case 'End':
        event.preventDefault();
        setFocusedIndex(items.length - 1);
        break;
        
      case 'Enter':
      case ' ':
        event.preventDefault();
        if (items[focusedIndex]) {
          onSelect(items[focusedIndex]);
        }
        break;
        
      case 'Escape':
        event.preventDefault();
        setFocusedId(null);
        break;
    }
  }, [items, focusedIndex, onSelect]);
  
  useEffect(() => {
    if (items[focusedIndex]) {
      setFocusedId(getItemId(items[focusedIndex]));
    }
  }, [focusedIndex, items, getItemId]);
  
  useEffect(() => {
    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [handleKeyDown]);
  
  return { focusedId, setFocusedIndex };
}

// Usage in components
export default function MasteryCriterionGrid() {
  const { focusedId, setFocusedIndex } = useKeyboardNavigation(
    criteria,
    handleCriterionSelect,
    criterion => criterion.id
  );
  
  return (
    <div className={styles.masteryGrid} role="grid" aria-label="Mastery criteria">
      {criteria.map((criterion, index) => (
        <div
          key={criterion.id}
          role="gridcell"
          tabIndex={focusedId === criterion.id ? 0 : -1}
          className={`${styles.criterionCard} ${focusedId === criterion.id ? styles.focused : ''}`}
          onClick={() => handleCriterionSelect(criterion)}
          onFocus={() => setFocusedIndex(index)}
          aria-label={`${criterion.title}, mastery level: ${criterion.masteryLevel}`}
        >
          {/* Criterion content */}
        </div>
      ))}
    </div>
  );
}
